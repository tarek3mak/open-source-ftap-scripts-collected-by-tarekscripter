                    TweenService:Create(Fill, TweenInfo.new(0.05), {Size = UDim2.new(pos, 0, 1, 0)}):Play()
                    pcall(Callback, val)
                end

                SliderFrame.InputBegan:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                        DraggingSlider = true
                        UpdateSlider(input)
                    end
                end)

                UserInputService.InputEnded:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                        DraggingSlider = false
                    end
                end)

                UserInputService.InputChanged:Connect(function(input)
                    if DraggingSlider and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                        UpdateSlider(input)
                    end
                end)

                if Flag then library.Flags[Flag] = SliderAPI end
                table.insert(TabObj.Elements, SliderFrame)
                return SliderAPI
            end

                local Options = Cfg.List or Cfg.Options or {}
                local CurrentOption = Cfg.Default or Cfg.CurrentOption or ""
                local Flag = Cfg.Flag
                local Callback = Cfg.Callback or function() end

                ListContainer.Parent = DropFrame

                local ListLayout = Instance.new("UIListLayout")
                ListLayout.SortOrder = Enum.SortOrder.LayoutOrder
                ListLayout.Padding = UDim.new(0, 3)
                ListLayout.Parent = ListContainer

                local DropdownAPI = { Value = CurrentOption, Callback = Callback }

                local function BuildOptions(list)
                    for _, child in ipairs(ListContainer:GetChildren()) do
                        if child:IsA("TextButton") then child:Destroy() end
                    end
                    Options = list or {}
                    for _, opt in ipairs(Options) do
                        local OptBtn = Instance.new("TextButton")
                        OptBtn.Size = UDim2.new(1, 0, 0, 22)
                        OptBtn.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
                        OptBtn.BackgroundTransparency = 0.94
                        OptBtn.Text = tostring(opt)
                        OptBtn.Font = Enum.Font.Gotham
                        OptBtn.TextColor3 = library.Theme.TextDark
                        OptBtn.TextSize = 10
                        OptBtn.TextTruncate = Enum.TextTruncate.AtEnd
                        OptBtn.Parent = ListContainer

                        local OptCorner = Instance.new("UICorner")
                        OptCorner.CornerRadius = UDim.new(0, 5)
                        OptCorner.Parent = OptBtn

                        OptBtn.MouseButton1Click:Connect(function()
                            CurrentOption = opt
                            DropdownAPI.Value = opt
                            Label.Text = Name .. ": " .. tostring(CurrentOption)
                            Expanded = false
                            Arrow.Text = "v"
                            TweenService:Create(DropFrame, TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Size = UDim2.new(1, 0, 0, 30)}):Play()
                            pcall(Callback, CurrentOption)
                        end)
                    end
                end

                BuildOptions(Options)

                function DropdownAPI:Refresh(RefreshCfg)
                    if type(RefreshCfg) == "table" then
                        if RefreshCfg.Text then Name = RefreshCfg.Text end
                        BuildOptions(RefreshCfg.List or RefreshCfg.Options or {})
                    end
                    Label.Text = Name
                    Expanded = false
                    Arrow.Text = "v"
                    DropFrame.Size = UDim2.new(1, 0, 0, 30)
                end

                ClickBtn.MouseButton1Click:Connect(function()
                    Expanded = not Expanded
                    Arrow.Text = Expanded and "^" or "v"
                    local targetHeight = Expanded and (34 + #Options * 25 + 4) or 30
                    TweenService:Create(DropFrame, TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Size = UDim2.new(1, 0, 0, targetHeight)}):Play()
                end)

                if Flag then library.Flags[Flag] = DropdownAPI end
                table.insert(TabObj.Elements, DropFrame)
                return DropdownAPI
            end

            function SectionObj:Keybind(Cfg)
                local Name = Cfg.Text or Cfg.Name or "Keybind"
                local Mode = Cfg.Mode or "Toggle"
                local CurrentKey = Cfg.Default or Enum.KeyCode.Unknown
                local Flag = Cfg.Flag
                local Callback = Cfg.Callback or function() end

                local Binding = false
                local KeyFrame = CreateGlassFrame(SectionBox, UDim2.new(1, 0, 0, 30))
                KeyFrame.BackgroundTransparency = library.Theme.ElementTransparency

                local Label = Instance.new("TextLabel")
                Label.Size = UDim2.new(1, -75, 1, 0)
                Label.Position = UDim2.new(0, 8, 0, 0)
                Label.BackgroundTransparency = 1
                Label.Font = Enum.Font.GothamMedium
                Label.Text = Name
                Label.TextColor3 = library.Theme.Text
                Label.TextSize = 11
                Label.TextTruncate = Enum.TextTruncate.AtEnd
                Label.TextXAlignment = Enum.TextXAlignment.Left
                Label.Parent = KeyFrame

                local KeyBtn = Instance.new("TextButton")
                KeyBtn.Size = UDim2.new(0, 62, 0, 20)
                KeyBtn.Position = UDim2.new(1, -68, 0.5, -10)
                KeyBtn.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
                KeyBtn.BackgroundTransparency = 0.5
                KeyBtn.Font = Enum.Font.GothamBold
                KeyBtn.Text = CurrentKey == Enum.KeyCode.Unknown and "None" or CurrentKey.Name
                KeyBtn.TextColor3 = library.Theme.Accent
                KeyBtn.TextSize = 10
                KeyBtn.Parent = KeyFrame

                local KeyCorner = Instance.new("UICorner")
                KeyCorner.CornerRadius = UDim.new(0, 5)
                KeyCorner.Parent = KeyBtn

                local KeyAPI = { Key = CurrentKey, Callback = Callback }

                KeyBtn.MouseButton1Click:Connect(function()
                    Binding = true
                    KeyBtn.Text = "..."
                end)

                UserInputService.InputBegan:Connect(function(input, gpe)
                    if Binding then
                        if input.UserInputType == Enum.UserInputType.Keyboard then
                            Binding = false
                            if input.KeyCode == Enum.KeyCode.Escape or input.KeyCode == Enum.KeyCode.Backspace then
                                CurrentKey = Enum.KeyCode.Unknown
                                KeyBtn.Text = "None"
                            else
                                CurrentKey = input.KeyCode
                                KeyBtn.Text = CurrentKey.Name
                            end
                            KeyAPI.Key = CurrentKey
                        end
                        return
                    end

                    if not gpe and CurrentKey ~= Enum.KeyCode.Unknown and input.KeyCode == CurrentKey then
                        if Mode == "Hold" then
                            pcall(Callback, true)
                        else
                            pcall(Callback)
                        end
                    end
                end)

                UserInputService.InputEnded:Connect(function(input)
                    if not Binding and Mode == "Hold" and CurrentKey ~= Enum.KeyCode.Unknown and input.KeyCode == CurrentKey then
                        pcall(Callback, false)
                    end
                end)

                if Flag then library.Flags[Flag] = KeyAPI end
                table.insert(TabObj.Elements, KeyFrame)
                return KeyAPI
            end
