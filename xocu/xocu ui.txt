local function CreateGlassFrame(parent, size, pos)
    local frame = Instance.new("Frame")
    frame.Size = size or UDim2.new(1, 0, 1, 0)
    frame.Position = pos or UDim2.new(0, 0, 0, 0)
    frame.BackgroundColor3 = Library.Theme.Background
    frame.BackgroundTransparency = Library.Theme.GlassyTransparency
    frame.BorderSizePixel = 0
    frame.Parent = parent

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 14)
    corner.Parent = frame

    local stroke = Instance.new("UIStroke")
    stroke.Color = Library.Theme.GlassyBorder
    stroke.Transparency = 0.7
    stroke.Thickness = 1.4
    stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    stroke.Parent = frame

    return frame
end

function Library:Notify(Config)
    local Title = Config.Title or "Notification"
    local Content = Config.Content or ""
    local Duration = Config.Duration or 3
    local IsError = Config.Error or false

    local Notif = CreateGlassFrame(NotificationsFrame, UDim2.new(1, 0, 0, 65))
    Notif.BackgroundTransparency = 1
    Notif.Position = UDim2.new(1, 320, 0, 0)

    local NotifStroke = Notif:FindFirstChildOfClass("UIStroke")
    local InitialStrokeTrans = NotifStroke and NotifStroke.Transparency or 0.7
    if NotifStroke then NotifStroke.Transparency = 1 end

    local AccentBar = Instance.new("Frame")
    AccentBar.Size = UDim2.new(0, 4, 1, -16)
    AccentBar.Position = UDim2.new(0, 8, 0, 8)
    AccentBar.BackgroundColor3 = IsError and Library.Theme.ErrorAccent or Library.Theme.Accent
    AccentBar.BorderSizePixel = 0
    AccentBar.BackgroundTransparency = 1
    AccentBar.Parent = Notif

    local BarCorner = Instance.new("UICorner")
    BarCorner.CornerRadius = UDim.new(1, 0)
    BarCorner.Parent = AccentBar

    local NTitle = Instance.new("TextLabel")
    NTitle.Size = UDim2.new(1, -30, 0, 20)
    NTitle.Position = UDim2.new(0, 20, 0, 10)
    NTitle.BackgroundTransparency = 1
    NTitle.Font = Enum.Font.GothamBold
    NTitle.Text = Title
    NTitle.TextColor3 = IsError and Library.Theme.ErrorAccent or Library.Theme.Text
    NTitle.TextTransparency = 1
    NTitle.TextSize = 13
    NTitle.TextXAlignment = Enum.TextXAlignment.Left
    NTitle.Parent = Notif

    local NDesc = Instance.new("TextLabel")
    NDesc.Size = UDim2.new(1, -30, 0, 26)
    NDesc.Position = UDim2.new(0, 20, 0, 30)
    NDesc.BackgroundTransparency = 1
    NDesc.Font = Enum.Font.Gotham
    NDesc.Text = Content
    NDesc.TextColor3 = Library.Theme.TextDark
    NDesc.TextTransparency = 1
    NDesc.TextSize = 11
    NDesc.TextWrapped = true
    NDesc.TextXAlignment = Enum.TextXAlignment.Left
    NDesc.Parent = Notif

    TweenService:Create(Notif, TweenInfo.new(0.45, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        Position = UDim2.new(0, 0, 0, 0),
        BackgroundTransparency = 0.1
    }):Play()

    if NotifStroke then
        TweenService:Create(NotifStroke, TweenInfo.new(0.45), {Transparency = InitialStrokeTrans}):Play()
    end
    TweenService:Create(AccentBar, TweenInfo.new(0.4), {BackgroundTransparency = 0}):Play()
    TweenService:Create(NTitle, TweenInfo.new(0.4), {TextTransparency = 0}):Play()
    TweenService:Create(NDesc, TweenInfo.new(0.4), {TextTransparency = 0}):Play()

    task.delay(Duration, function()
        local tweenOut = TweenService:Create(Notif, TweenInfo.new(0.35, Enum.EasingStyle.Quart, Enum.EasingDirection.In), {
            Position = UDim2.new(1, 320, 0, 0),
            BackgroundTransparency = 1
        })
        if NotifStroke then TweenService:Create(NotifStroke, TweenInfo.new(0.35), {Transparency = 1}):Play() end
        TweenService:Create(AccentBar, TweenInfo.new(0.3), {BackgroundTransparency = 1}):Play()
        TweenService:Create(NTitle, TweenInfo.new(0.3), {TextTransparency = 1}):Play()
        TweenService:Create(NDesc, TweenInfo.new(0.3), {TextTransparency = 1}):Play()
        
        tweenOut:Play()
        tweenOut.Completed:Connect(function()
            Notif:Destroy()
        end)
    end)
end

                    local targetColor = CurrentValue and Library.Theme.Accent or Color3.fromRGB(40, 40, 50)

                    local val = math.floor(Min + (Max - Min) * pos)
                    ValLabel.Text = tostring(val)
                    Fill.Size = UDim2.new(pos, 0, 1, 0)
                    if Flag then Library.Flags[Flag] = val end
                    pcall(Callback, val)
                end

                SliderFrame.InputBegan:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                        DraggingSlider = true
                        UpdateSlider(input)
                    end
                end)

                UserInputService.InputEnded:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                        DraggingSlider = false
                    end
                end)

                UserInputService.InputChanged:Connect(function(input)
                    if DraggingSlider and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                        UpdateSlider(input)
                    end
                end)
            end
